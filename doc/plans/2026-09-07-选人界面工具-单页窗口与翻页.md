## Changes

### 1. pplyCanvasSize() 重构 —— 物理画布永远 800×600

`js
function applyCanvasSize() {
  const VP_W = 800, VP_H = 600;
  const W = Math.max(1, state.layout.width || VP_W);
  const H = Math.max(1, state.layout.height || VP_H);
  const scale = Math.min(VP_W / W, VP_H / H);
  state.viewport = {
    vpW: VP_W, vpH: VP_H,
    srcW: W, srcH: H,
    scale: scale,
    drawW: W * scale, drawH: H * scale,
    offX: Math.round((VP_W - W * scale) / 2),
    offY: Math.round((VP_H - H * scale) / 2),
  };
  canvas.width = VP_W; canvas.height = VP_H;
  canvas.style.width = VP_W + 'px';
  canvas.style.height = VP_H + 'px';
}
`

### 2. 坐标映射 helper

`js
function mapToScreen(x, y) {
  const v = state.viewport;
  return { x: x * v.scale + v.offX, y: y * v.scale + v.offY };
}
function mapFromScreen(x, y) {
  const v = state.viewport;
  return { x: (x - v.offX) / v.scale, y: (y - v.offY) / v.scale };
}
`

### 6. PAGE_HEIGHT 可配置 + currentPage

`js
state.PAGE_HEIGHT = 600;     // 默认
state.currentPage = 0;

function totalPages() {
  const H = state.layout.height || 600;
  return Math.max(1, Math.ceil(H / state.PAGE_HEIGHT));
}

function gotoPage(n) {
  state.currentPage = Math.max(0, Math.min(totalPages() - 1, n));
  render();
  renderPageIndicator();
  snapMainScrollToPage();
}
`

### 8. keydown handler 新增 ↑↓/PageUp/PageDown/Home/End

`js
window.addEventListener('keydown', e => {
  const ae = document.activeElement;
  const inInput = ae && (ae.tagName === 'INPUT' || ae.tagName === 'BUTTON' || ae.tagName === 'SELECT');

  if (e.key === 'ArrowDown' || e.key === 'PageDown') {
    if (inInput) return;  // 现有微调逻辑接管
    e.preventDefault(); gotoPage(state.currentPage + 1); return;
  }
  if (e.key === 'ArrowUp' || e.key === 'PageUp') {
    if (inInput) return;
    e.preventDefault(); gotoPage(state.currentPage - 1); return;
  }
  if (e.key === 'Home') { e.preventDefault(); gotoPage(0); return; }
  if (e.key === 'End')  { e.preventDefault(); gotoPage(totalPages() - 1); return; }
  // ... 现有 Ctrl+Z / Ctrl+S / Esc / Delete 不动
});
`

### 9. 滚轮 + 滚动条 snap 到页

`js
main.addEventListener('scroll', () => {
  const y = main.scrollTop;
  const page = Math.round(y / state.PAGE_HEIGHT);
  const target = page * state.PAGE_HEIGHT;
  if (Math.abs(y - target) > 1) {
    main.scrollTo({ top: target, behavior: 'smooth' });
  }
  if (page !== state.currentPage) {
    state.currentPage = page;
    renderPageIndicator();
  }
});
`

## Risks

| Risk | Mitigation |
|---|---|
| verify-canvas-size.js 断言 canvas.width=layout.width 会失败 | 更新断言: canvas.width=800 (视口固定); scale/drawW/drawH 由 layout 算 |
| 缩放+反算误差 → 拖动精度 | 用 Math.round 写入 offset; scale 精度 4 位小数 |
| undo 跨页 → currentPage 错位 | snapshot 加 currentPage + PAGE_HEIGHT |
| PageUp/Down 与微调冲突 | inInput 检测 |
| 解析素材 840 宽 + 360px 侧栏在低分辨率屏挤压 | main min-width: 800 提示; 用户可缩小浏览器 |
